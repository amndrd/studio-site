27  async charger(url = '/modeles/one.glb') {
28    const gltf = await new GLTFLoader().loadAsync(url)
29    gltf.scene.traverse((o) => {
30      if (!(o instanceof THREE.Mesh)) return
31      if (o.name === 'tissu') this.tissu = o.material
32      if (o.name === 'plateau') this.plateau = o.material
33      o.castShadow = o.receiveShadow = true
34    })
35    this.scene.add(gltf.scene)
36    this.appliquer(teintes.sauge)
37  }

39  appliquer(t: Teinte) {

43  }
